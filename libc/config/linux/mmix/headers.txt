set(TARGET_PUBLIC_HEADERS
    libc.include.errno
    libc.include.features
    libc.include.limits
    # Public types/declarations do not enable signal or context providers.
    libc.include.signal
    libc.include.setjmp
    libc.include.ucontext
    libc.include.sys_ucontext
    libc.include.stdint
    libc.include.stdlib
    libc.include.string
    # Required by the generated string.h compatibility include.
    libc.include.strings
    libc.include.sys_mman
    libc.include.sys_syscall
    libc.include.sys_types
    libc.include.unistd
)

  list(APPEND TARGET_PUBLIC_HEADERS libc.include.ctype libc.include.inttypes
    libc.include.math)
endif()

  list(APPEND TARGET_PUBLIC_HEADERS libc.include.fcntl)
endif()
if(LIBC_MMIX_BUILD_STDIO)
  list(APPEND TARGET_PUBLIC_HEADERS libc.include.stdio)
endif()

  list(APPEND TARGET_PUBLIC_HEADERS libc.include.wchar libc.include.wctype)
endif()
if(LIBC_MMIX_BUILD_FILESYSTEM)
  list(APPEND TARGET_PUBLIC_HEADERS libc.include.sys_stat libc.include.dirent
    libc.include.stdio libc.include.pwd libc.include.sys_vfs libc.include.sys_statfs)
endif()
if(LIBC_MMIX_BUILD_TIME)
  list(APPEND TARGET_PUBLIC_HEADERS libc.include.time libc.include.sys_time)
endif()
if(LIBC_MMIX_BUILD_PROCESS_BASE)
  # LLVM's common Unix headers include dlfcn.h without using loader services.

    libc.include.dlfcn libc.include.sysexits)
endif()
if(LIBC_MMIX_BUILD_SPAWN)
  list(APPEND TARGET_PUBLIC_HEADERS libc.include.spawn)
endif()

  )
